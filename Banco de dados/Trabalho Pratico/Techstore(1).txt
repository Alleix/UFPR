Create Table Produto(
  Produto_id int primary key,
  Nome varchar(30),
  Descricao varchar(100),
  Preco float,
  Categoria varchar(30)
  );

Create Table Clientes (
  Cliente_id int primary key,
  Nome varchar(30),
  Endereco varchar(100),
  Email varchar(100),
  Telefone varchar(30)

);

Create Table Pedido (
  id_pedido int primary key,
  fk_Produto_id int ,
  foreign key (fk_Produto_id) references Produto(Produto_id),
  fk_Cliente_id int ,
  foreign key (fk_Cliente_id) references clientes(Cliente_id),
  DataPedido varchar(30),
  Quantidade int
);


Create Table Estoque (
  Estoque_id int primary key,
  Quantidade int,
  Localizacao_Estq varchar(100),
  Validade varchar(30)
);

Create Table ProdutoEstoque (
fk_Produto_id int,
foreign key (fk_Produto_id) references Produto(Produto_id),
fk_Estoque_id int,
foreign key (fk_Estoque_id) references Estoque(Estoque_id)
);
